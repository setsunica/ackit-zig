const std = @import("std");

fn solve(
    allocator: std.mem.Allocator,
    scanner: *ackit.io.Scanner,
    printer: *ackit.io.Printer,
) !void {
    const parsed = try scanner.scanAllAlloc([]const u8, allocator);
    defer parsed.deinit();
    const input = parsed.value;
    try printer.print("{f}", .{input});
}

pub fn main() !void {
    var sfa = std.heap.stackFallback(4096, std.heap.page_allocator);
    try ackit.io.interactStdio(sfa.get(), solve);
}
